// 1. Single selection mode (default)
const file = await MBKBucket.openGallery({
  prefix: 'images/', // start folder
});
if (file) console.log(file.key, file.name, file.size, file.url);

// 2. Multi-selection mode with In-Modal Upload
const files = await MBKBucket.openGallery({
  multiple: true,
  maxFiles: 10,
});
if (files) {
  // returns Array<{ key, name, size, url }>
  files.forEach(f => console.log(f.key));
}

const picker = MBKBucket.createPicker('#avatar-input', {
  preview:     '#avatar-live-img', // Selector for <img> or background container
  prefix:      'avatars/',
  multiple:    false,
  placeholder: 'Choose an avatar…',
  onSelect: result => {
    console.log('Selected:', result?.key, result?.url);
  },
});

// Mount in any container element
const uploader = MBKBucket.createUploader('#my-container', {
  prefix:   'uploads/',
  bucket:   '',
  accept:   'image/*,.pdf',
  multiple: true,
  onSuccess: ({ file, key }) => console.log('Uploaded:', key),
  onError:   ({ file, error }) => console.error(error.message),
});

// Programmatic injection
uploader.addFiles(fileInput.files);
uploader.clear();

// List files
